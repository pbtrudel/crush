```ts
// toScreen turns a projected node into a tag anchor. A degenerate camera
// (a zero-size canvas, a NaN pose) can project NaN, and NaN never equals
// itself, so such a node is reported hidden at the origin: the frame then
// compares equal to the next one and nothing republishes every frame.
export function toScreen(id: string, ndc: { x: number; y: number; z: number }, eyeZ: number, width: number, height: number): AgentScreen {
  const x = Math.round(((ndc.x + 1) / 2) * width);
  const y = Math.round(((1 - ndc.y) / 2) * height);
  if (!Number.isFinite(x) || !Number.isFinite(y) || !Number.isFinite(ndc.z) || !Number.isFinite(eyeZ)) {
    return { id, x: 0, y: 0, visible: false };
  }
```

```go
func (c *coordinator) beginSubSession(ctx context.Context, sessionID string) (context.Context, func()) {

		c.subSessionsMu.Unlock()

	}
```

```go
	sessionID := c.sessions.CreateAgentToolSessionID(params.AgentMessageID, params.ToolCallID)
	// Cancellable before it is listed: the browser can click cancel the
	// moment agents.json shows the run.
	ctx, release := c.beginSubSession(ctx, sessionID)

	sess, err := c.createSubAgentSession(ctx, sessionID, params)

	// as failed instead of leaving it listed as running.

	done := c.trackSubSession(sess.ID, params.Agent)
	// Deferred: a panic in the run must not leave the child session busy.

	result, err := params.Agent.Run(ctx, call)
	runErr = err
	c.askToSignInAgain(err, model)
	if err != nil {
		return subAgentRunError("Failed to generate response", err), nil
	}
	return subAgentResponse(result), nil
```

```go
package web

import "context"

// maxConcurrentDiffs bounds the diff work of every tab together. One diff
// of two 8 MiB sides can hold a few hundred megabytes while it runs, so
// three tabs refreshing at once must queue instead of stacking up.
const maxConcurrentDiffs = 2

// diffSlots is a counting semaphore for diff work. A request whose client
// went away stops waiting.
type diffSlots struct{ ch chan struct{} }

func newDiffSlots(n int) *diffSlots { return &diffSlots{ch: make(chan struct{}, n)} }

func (d *diffSlots) acquire(ctx context.Context) (func(), error) {
	select {
	case d.ch <- struct{}{}:
		return func() { <-d.ch }, nil
	case <-ctx.Done():
		return nil, ctx.Err()
	}
}
```

In `files.go`:

```go
	// maxDiffInputLines bounds one side of a diff by lines. The diff splits
	// each side into a slice of lines, so a side of short lines costs far
	// more than its bytes; 200,000 lines keep one request's split under
	// about 7 MB of slice headers per side.
	maxDiffInputLines = 200_000
```

- `treeHistory` lists `ListSessionHistoryMeta` and sorts `[]history.FileMeta`

```go
import (
	"errors"

	"github.com/charmbracelet/crush/internal/workspace"
```

```go
// agentStartingText is what the browser shows while the coder agent is not
// ready yet: no model chosen, or the coordinator being rebuilt after a
// config reload.
const agentStartingText = "Crush is starting the agent"

// writeWorkspaceError answers a failed workspace call. An agent that is
// not ready yet is a state the caller can wait out, so it is 503 with a
// code the browser shows as "starting"; anything else is a 500 without the
// error text.
func writeWorkspaceError(w http.ResponseWriter, code string, err error) {
	if errors.Is(err, workspace.ErrAgentNotInitialized) {
		writeError(w, http.StatusServiceUnavailable, "agent_starting", agentStartingText)
		return
	}
	writeInternal(w, code, err)
}

// promptFailure is the web_error payload for a detached prompt that could
// not run. The error text is fixed, never the workspace error, which can
// name files or sockets; the log keeps the detail.
func promptFailure(sessionID string, err error) map[string]string {
	if errors.Is(err, workspace.ErrAgentNotInitialized) {
		return map[string]string{"session_id": sessionID, "code": "agent_starting", "error": agentStartingText + "; send the prompt again in a moment"}
	}
	return map[string]string{"session_id": sessionID, "code": "prompt_failed", "error": "The prompt could not run. The terminal log has the details."}
}
```

```go
func (g *Gateway) handleRenameSession(w http.ResponseWriter, r *http.Request) {
	var body struct {
		Title string `json:"title"`
	}
	if !decodeJSON(w, r, &body) {
		return
	}
	title := strings.TrimSpace(body.Title)
	if title == "" || utf8.RuneCountInString(title) > maxTitleRunes {
		writeError(w, http.StatusBadRequest, "bad_title", "title must be 1 to 200 characters")
		return
	}
	sess, err := g.ws.GetSession(r.Context(), r.PathValue("sid"))
	if err != nil {
		if isNotFound(err) {
			writeError(w, http.StatusNotFound, "unknown_session", "no such session")
			return
		}
		writeWorkspaceError(w, "get_session", err)
		return
	}

	saved, err := g.ws.SaveSession(r.Context(), sess)
	if err != nil {
		writeWorkspaceError(w, "save_session", err)
		return
	}
	writeJSON(w, http.StatusOK, innerPayload(pubsub.Event[session.Session]{Type: pubsub.UpdatedEvent, Payload: saved}))
}
```

```go
// Substitute replaces each $NAME placeholder in content with its argument
// in a single pass. A value is inserted as is and never scanned again, so
// an argument that contains "$OTHER" cannot pull in another argument, and
// the result does not depend on map order. Placeholders without an
// argument are left in place.
func Substitute(content string, args map[string]string) string {
	return namedArgPattern.ReplaceAllStringFunc(content, func(m string) string {
		if v, ok := args[m[1:]]; ok {
			return v
		}
		return m
	})
}
```

  it, so M3 adds no `tool_call_id` column. Carried to M4.
- **Unpin to browse another session (spec 2.5).** The HUD switcher
  switches the terminal (`POST /follow`), like the TUI sessions dialog;
  the tab always follows the terminal. Browsing a different session than
  the terminal's would split the prompt target and the permission routing
  between the two clients. Carried to M4.
- **Palette scope.** The palette covers every TUI command that acts on
  the session, the model, MCP, LSP, skills or custom commands (Task 17's
  list). Terminal-only commands are left out because the browser has no
  equivalent or has its own control: Toggle Sidebar, Toggle Help, Toggle
  Mouse, Toggle Transparent Background, Themes (M4), Open External
  Editor, Open File Picker (the deck's attach button), Toggle Secret
  Reveal, Open Web Mode, Quit, Toggle Pills, View Agents (the agent list

  first are carried to M4: Export Session as Markdown, Set Goal, Start
  Loop, Toggle Agent Teams, Enable and Disable Docker MCP Catalog,
  Channels, Toggle Remote Control (Discord), Notifications settings.
- **Config routes.** Spec 2.6 lists `GET /config` and
  `POST /config/{set,remove,reload}`. M3 surfaces need only the model and
  reasoning fields (Task 13) and the reload action. Arbitrary `set` and
  `remove` from the browser would let a page write any config key; they
  are carried to M4 with the visual settings that need them. Reload is in
  M3 as `POST /api/config/reload` (Task 14), spelled under `/config` as

  enabled providers only. Choosing an unconfigured provider needs a sign-in
